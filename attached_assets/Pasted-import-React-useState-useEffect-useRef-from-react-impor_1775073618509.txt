// --- Safe Local Storage Helper ---
const safeGetStorage = (key, defaultValue = '') => {
  try {
    return localStorage.getItem(key) || defaultValue;
  } catch (e) {
    return defaultValue;
  }
};

const safeSetStorage = (key, value) => {
  try {
    localStorage.setItem(key, value);
  } catch (e) {
    console.warn("Local storage is disabled.");
  }
};

// --- Matrix Rain Component ---
const MatrixRain = ({ opacityClass = "opacity-20" }) => {
  const canvasRef = useRef(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    
    canvas.width = window.innerWidth;
    canvas.height = window.innerHeight;
    
    const katakana = 'アァカサタナハマヤャラワガザダバパイィキシチニヒミリヰギジヂビピウゥクスツヌフムユュルグズブヅプエェケセテネヘメレゲゼデベペオォコソトノホモヨョロゴゾドボポヴッン';
    const latin = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ';
    const nums = '0123456789';
    const alphabet = katakana + latin + nums;
    
    const fontSize = 14;
    const columns = canvas.width / fontSize;
    
    const drops = [];
    for (let x = 0; x < columns; x++) {
      drops[x] = Math.random() * -100;
    }
    
    const draw = () => {
      ctx.fillStyle = 'rgba(5, 5, 5, 0.05)';
      ctx.fillRect(0, 0, canvas.width, canvas.height);
      
      ctx.fillStyle = '#00e5ff';
      ctx.font = fontSize + 'px monospace';
      
      for (let i = 0; i < drops.length; i++) {
        const text = alphabet.charAt(Math.floor(Math.random() * alphabet.length));
        
        if (drops[i] > 0) {
           ctx.fillText(text, i * fontSize, drops[i] * fontSize);
        }
        
        if (drops[i] * fontSize > canvas.height && Math.random() > 0.975) {
          drops[i] = 0;
        }
        drops[i]++;
      }
    };
    
    const interval = setInterval(draw, 33);
    
    const handleResize = () => {
      canvas.width = window.innerWidth;
      canvas.height = window.innerHeight;
    };
    
    window.addEventListener('resize', handleResize);
    return () => {
      clearInterval(interval);
      window.removeEventListener('resize', handleResize);
    };
  }, []);

  return (
    <canvas 
      ref={canvasRef} 
      className={`fixed top-0 left-0 w-full h-full z-0 pointer-events-none transition-opacity duration-1000 ${opacityClass}`} 
    />
  );
};

// --- Glowing Fireflies Component (ပိုးစုန်းကြူးများ) ---
const Fireflies = () => {
  const [flies, setFlies] = useState([]);

  useEffect(() => {
    // Generate 25 random fireflies
    const newFlies = Array.from({ length: 25 }).map((_, i) => ({
      id: i,
      left: `${Math.random() * 100}vw`,
      top: `${Math.random() * 100}vh`,
      animationDuration: `${Math.random() * 4 + 2}s`, 
      animationDelay: `${Math.random() * 5}s`,
      moveDuration: `${Math.random() * 20 + 15}s`,
      color: Math.random() > 0.3 ? '#00ff00' : '#00e5ff'
    }));
    setFlies(newFlies);
  }, []);

  return (
    <div className="fixed inset-0 z-0 pointer-events-none overflow-hidden">
      {flies.map((fly) => (
        <div
          key={fly.id}
          className="absolute rounded-full"
          style={{
            left: fly.left,
            top: fly.top,
            width: '3px',
            height: '3px',
            backgroundColor: fly.color,
            boxShadow: `0 0 10px 3px ${fly.color}`,
            animation: `firefly-blink ${fly.animationDuration} infinite ease-in-out ${fly.animationDelay}, firefly-float ${fly.moveDuration} infinite linear ${fly.animationDelay}`
          }}
        />
      ))}
    </div>
  );
};
